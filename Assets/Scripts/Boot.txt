
using UnityEngine;


[DefaultExecutionOrder(-1000)]  // runs very early
public class AppBootstrap : MonoBehaviour
{
    [SerializeField] private bool startInMainMenu = true;
    private static bool _made;

    void Awake()
    {
        Debug.Log("Starting the App Bootstrap");
        if (_made) { Destroy(gameObject); return; }
        _made = true;

        DontDestroyOnLoad(gameObject);

        // Create services
        Debug.Log("Creating Services");
        var lobby = new LobbyService();
        var game = new GameService();
        var flow = new GameFlow(lobby, game);

        // Register for global access
        Debug.Log("Registering services for Global Access in the Service Locator");
        ServiceLocator.Lobby = lobby;
        ServiceLocator.Game = game;
        ServiceLocator.GameFlow = flow;
        // Optional: force a starting phase once GameFlow exists.
        if (startInMainMenu)                                                    // Should we start in the main menu?
            flow.BackToMenu();                                                  //   -> Jump to MainMenu phase
        Debug.Log("[AppBootstrap] GameFlow created");                           // Log that GameFlow is ready

        // Create the shared RobotDirectory instance used by the whole app.
        ServiceLocator.RobotDirectory = new RobotDirectory();                   // Allocate robot registry
        Debug.Log($"[AppBootstrap] RobotDirectory created: {ServiceLocator.RobotDirectory.GetHashCode()}"); // Log instance id

        // Create ONE shared PlayersService and seed it with defaults.
        var players = new PlayersService();                                     // Allocate players service
        players.EnsureDefaults();                                               // Make sure Player1 / Player2 exist
        ServiceLocator.Players = players;                                       // Expose via ServiceLocator
        Debug.Log($"[AppBootstrap] PlayersService created: {players.GetHashCode()} with {players.GetAll().Count} players"); // Log


    }
}


public static class ServiceLocator
{
    public static IRobotDirectory RobotDirectory; 
    public static GameFlow GameFlow;
    public static LobbyService Lobby;
    public static GameService Game;
    public static RobotWebSocketServer RobotServer;  // Set by RobotWebSocketServer when it starts
    public static PlayersService Players;

}
