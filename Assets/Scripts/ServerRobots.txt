using System;                                     // For Action delegates
using System.Collections.Generic;                 // For List<T>
using TMPro;                                      // For TMP_InputField, TMP_Dropdown
using UnityEngine;                                // For MonoBehaviour, GameObject
using UnityEngine.UI;                             // For Button

// RenamePopup — small modal to edit a robot's name and assigned player.
// NOTE: Supports BOTH a legacy 2-arg callback and a new 3-arg callback.
//   - Legacy:  Action<string /*robotId*/, string /*playerOrNull*/>
//   - New:     Action<string /*robotId*/, string /*newName*/, string /*playerOrNull*/>
public class RenamePopup : MonoBehaviour
{
    [SerializeField] private TMP_InputField nameInput;     // Text box for the robot name
    [SerializeField] private TMP_Dropdown playerDropdown; // Dropdown of players (index 0 = Unassigned)
    [SerializeField] private Button okButton;      // OK button
    [SerializeField] private Button cancelButton;  // Cancel button

    // The robot we are editing (stable id).
    private string _robotId;

    // LEGACY: callback used by older callers (id, playerOrNull).
    private Action<string, string> _onApplyLegacy;

    // NEW: callback that also receives the new name (id, newName, playerOrNull).
    private Action<string, string, string> _onApplyWithName;

    // Cached PlayersService for names, kept very simple.
    private PlayersService _players;

    // Guard to suppress UI callbacks while we update programmatically.
    private bool _updating = false;

    // ------------------------------------------------------------------------
    // PUBLIC API (NEW preferred)
    // ------------------------------------------------------------------------

    // Preferred entry: open with explicit fields and the NEW 3-arg callback.
    public void Show(string robotId, string currentName, string currentAssignedPlayer, Action<string, string, string> onApply)
    {
        // Store robot id for the callback.
        _robotId = robotId;                                        // Remember id

        // Store the NEW callback (name + player).
        _onApplyWithName = onApply;                                // Set new callback

        // Clear the legacy callback to avoid ambiguity.
        _onApplyLegacy = null;                                     // Ensure legacy is null

        // Ensure PlayersService exists and cache it.
        if (ServiceLocator.Players == null) ServiceLocator.Players = new PlayersService(); // Create if needed
        _players = ServiceLocator.Players;                          // Cache reference

        // Subscribe to players changed so our dropdown updates live.
        _players.OnChanged -= HandlePlayersChanged;                 // Avoid double-subscribe
        _players.OnChanged += HandlePlayersChanged;                 // Subscribe

        // Wire buttons once (idempotent).
        okButton.onClick.RemoveListener(OnOkClicked);              // Unhook just in case
        okButton.onClick.AddListener(OnOkClicked);                 // Hook OK
        cancelButton.onClick.RemoveListener(OnCancelClicked);      // Unhook just in case
        cancelButton.onClick.AddListener(OnCancelClicked);         // Hook Cancel

        // Build UI under suppression.
        _updating = true;                                          // Start suppression
        BuildPlayersDropdown();                                     // Fill dropdown from service
        nameInput.SetTextWithoutNotify(currentName ?? "");          // Pre-fill name field
        SelectPlayerByNameWithoutNotify(currentAssignedPlayer);     // Pre-select player or Unassigned
        _updating = false;                                         // End suppression

        // Show the popup.
        gameObject.SetActive(true);                                 // Display modal
    }

    // Convenience alias: Open(...) for NEW signature (keeps nice callsites).
    public void Open(string robotId, string currentName, string currentAssignedPlayer, Action<string, string, string> onApply)
    {
        // Forward to Show(...) with the same signature.
        Show(robotId, currentName, currentAssignedPlayer, onApply); // Delegate
    }

    // ------------------------------------------------------------------------
    // PUBLIC API (LEGACY compatibility)
    // ------------------------------------------------------------------------

    // Legacy entry: open with the OLD 2-arg callback (id, playerOrNull).
    // Callers using this will NOT receive the new name via callback; if you need the name,
    // switch your call to the NEW 3-arg overload above.
    public void Show(string robotId, string currentName, string currentAssignedPlayer, Action<string, string> onApplyLegacy)
    {
        // Store robot id for callback.
        _robotId = robotId;                                        // Remember id

        // Store LEGACY callback (player only).
        _onApplyLegacy = onApplyLegacy;                            // Set legacy callback

        // Clear the new callback so we know which one to use at OK time.
        _onApplyWithName = null;                                   // Ensure new is null

        // Ensure PlayersService exists and cache it.
        if (ServiceLocator.Players == null) ServiceLocator.Players = new PlayersService(); // Create if needed
        _players = ServiceLocator.Players;                          // Cache reference

        // Subscribe to players changed so our dropdown updates live.
        _players.OnChanged -= HandlePlayersChanged;                 // Avoid double-subscribe
        _players.OnChanged += HandlePlayersChanged;                 // Subscribe

        // Wire buttons.
        okButton.onClick.RemoveListener(OnOkClicked);              // Unhook just in case
        okButton.onClick.AddListener(OnOkClicked);                 // Hook OK
        cancelButton.onClick.RemoveListener(OnCancelClicked);      // Unhook just in case
        cancelButton.onClick.AddListener(OnCancelClicked);         // Hook Cancel

        // Build UI under suppression.
        _updating = true;                                          // Start suppression
        BuildPlayersDropdown();                                     // Fill dropdown from service
        nameInput.SetTextWithoutNotify(currentName ?? "");          // Pre-fill name field
        SelectPlayerByNameWithoutNotify(currentAssignedPlayer);     // Pre-select player or Unassigned
        _updating = false;                                         // End suppression

        // Show the popup.
        gameObject.SetActive(true);                                 // Display modal
    }

    // Legacy alias: Open(...) for OLD signature.
    public void Open(string robotId, string currentName, string currentAssignedPlayer, Action<string, string> onApplyLegacy)
    {
        // Forward to legacy Show(...).
        Show(robotId, currentName, currentAssignedPlayer, onApplyLegacy); // Delegate
    }

    // Optional: Open directly from a RobotInfo using NEW callback (id, newName, playerOrNull).
    public void Open(RobotInfo info, Action<string, string, string> onApply)
    {
        // Guard null.
        if (info == null) return;                                   // Nothing to show

        // Extract current values with safe fallbacks.
        string id = info.RobotId;                                 // Stable id
        string name = string.IsNullOrEmpty(info.Callsign) ? info.RobotId : info.Callsign; // Show callsign or id
        string assignedPlayer = string.IsNullOrEmpty(info.AssignedPlayer) ? null : info.AssignedPlayer; // Null if none

        // Forward to NEW Show(...).
        Show(id, name, assignedPlayer, onApply);                    // Delegate
    }

    // ------------------------------------------------------------------------
    // Close / cleanup
    // ------------------------------------------------------------------------

    // Hide the popup and unhook events.
    public void Hide()
    {
        // Unsubscribe from players changes to avoid leaks.
        if (_players != null) _players.OnChanged -= HandlePlayersChanged; // Unhook

        // Hide the modal window.
        gameObject.SetActive(false);                                 // Close
    }

    // ------------------------------------------------------------------------
    // Internals: building and reading the dropdown
    // ------------------------------------------------------------------------

    // Rebuild the players dropdown from PlayersService.
    private void BuildPlayersDropdown()
    {
        // Start with an "Unassigned" option at index 0.
        var options = new List<TMP_Dropdown.OptionData>();           // New list
        options.Add(new TMP_Dropdown.OptionData("Unassigned"));      // Index 0

        // Snapshot players list (or empty).
        var list = _players != null ? _players.GetAll() : new List<PlayerInfo>(); // Copy

        // Append each player's Name (fallback label if missing).
        for (int i = 0; i < list.Count; i++)
        {
            string label = string.IsNullOrEmpty(list[i].Name) ? $"Player{i + 1}" : list[i].Name; // Safe label
            options.Add(new TMP_Dropdown.OptionData(label));           // Add option
        }

        // Replace dropdown options without firing callbacks.
        playerDropdown.ClearOptions();                                 // Remove old
        playerDropdown.AddOptions(options);                            // Add new
        playerDropdown.SetValueWithoutNotify(0);                       // Default to Unassigned
        playerDropdown.RefreshShownValue();                            // Refresh caption
    }

    // Programmatically select a player name in the dropdown (null/unknown -> Unassigned).
    private void SelectPlayerByNameWithoutNotify(string playerNameOrNull)
    {
        // If blank, select Unassigned.
        if (string.IsNullOrEmpty(playerNameOrNull))
        {
            playerDropdown.SetValueWithoutNotify(0);                  // Unassigned
            playerDropdown.RefreshShownValue();                       // Refresh
            return;                                                   // Done
        }

        // Look for a matching option (skip index 0).
        for (int i = 1; i < playerDropdown.options.Count; i++)
        {
            if (string.Equals(playerDropdown.options[i].text, playerNameOrNull, StringComparison.Ordinal))
            {
                playerDropdown.SetValueWithoutNotify(i);              // Select by name
                playerDropdown.RefreshShownValue();                   // Refresh
                return;                                               // Done
            }
        }

        // If not found, fall back to Unassigned.
        playerDropdown.SetValueWithoutNotify(0);                      // Default
        playerDropdown.RefreshShownValue();                           // Refresh
    }

    // Read the selected player's name (returns null if Unassigned).
    private string GetSelectedPlayerNameOrNull()
    {
        // Index 0 means Unassigned.
        if (playerDropdown.value == 0) return null;                   // No assignment

        // Otherwise return the selected label text.
        return playerDropdown.options[playerDropdown.value].text;     // Player name
    }

    // ------------------------------------------------------------------------
    // Event reactions
    // ------------------------------------------------------------------------

    // PlayersService changed (add/remove/rename/alliance) ? rebuild dropdown, preserve selection.
    private void HandlePlayersChanged()
    {
        // Do rebuild under suppression to avoid UI loops.
        _updating = true;                                            // Begin suppression
        string selected = GetSelectedPlayerNameOrNull();             // Remember current choice
        BuildPlayersDropdown();                                       // Rebuild options
        SelectPlayerByNameWithoutNotify(selected);                    // Restore selection if possible
        _updating = false;                                           // End suppression
    }

    // OK: invoke whichever callback is set, passing NEW NAME (and player).
    private void OnOkClicked()
    {
        // Ignore if we’re in a programmatic update.
        if (_updating) return;                                       // Guard

        // Read the edited name (trim whitespace).
        string newName = nameInput != null ? nameInput.text.Trim() : ""; // Edited name

        // Read selected player (null if Unassigned).
        string playerOrNull = GetSelectedPlayerNameOrNull();         // Player name or null

        // Prefer NEW callback if provided (id, newName, playerOrNull).
        if (_onApplyWithName != null)
        {
            _onApplyWithName.Invoke(_robotId, newName, playerOrNull); // Send all data
        }
        else
        {
            // Fallback: legacy callback gets only (id, playerOrNull).
            // NOTE: The caller WON'T receive the new name in this path.
            _onApplyLegacy?.Invoke(_robotId, playerOrNull);           // Legacy usage
        }

        // Close the popup.
        Hide();                                                       // Done
    }

    // Cancel: close without invoking any callback.
    private void OnCancelClicked()
    {
        // Ignore if we’re in a programmatic update.
        if (_updating) return;                                       // Guard
        Hide();                                                      // Close
    }
}
using System;                          // Basic C# types like Action
using System.Collections.Generic;      // Lists
using UnityEngine;                     // Unity types (MonoBehaviour, GameObject)
using UnityEngine.UI;                  // UI Button
using TMPro;                           // TextMeshPro

// Attach this script to your "Server > Robots" panel.
// Drag the ScrollView's Content into "content" in the Inspector.
// Drag your row prefab into "rowPrefab".
// (Optional) Drag a RenamePopup into "renamePopup".
public class RobotsPanelPresenter : MonoBehaviour
{
    // === Inspector fields ===

    [Header("ScrollView wiring")]
    [SerializeField] private RectTransform content;   // Where the rows live under the ScrollView
    [SerializeField] private GameObject rowPrefab;    // Prefab for one robot row

    [Header("Optional: small modal to rename/assign")]
    [SerializeField] private RenamePopup renamePopup; // Can be null if you haven't made one yet

    // === Private fields ===

    private IRobotDirectory _dir;                     // The shared robot registry (from ServiceLocator)
    private bool _isSubscribed = false;               // Tracks if we subscribed to events already (safety)

    // Called automatically when this component becomes enabled/visible
    private void OnEnable()
    {
        // Grab the shared directory from the global shelf
        _dir = ServiceLocator.RobotDirectory;

        // Log which directory instance we are talking to (helps spot duplicate registries)
        // GetHashCode() just gives us a number to compare in the Console.
        Debug.Log($"[RobotsPanelPresenter:{name}] OnEnable - Using RobotDirectory hash={_dir?.GetHashCode()}");

        // If the directory is missing, we can't continue. Usually means AppBootstrap didn't run.
        if (_dir == null)
        {
            Debug.LogError("[RobotsPanelPresenter] RobotDirectory is null. Is AppBootstrap in the scene and enabled?");
            return; // Bail out early to avoid null errors
        }

        // (Safety) Make sure content and rowPrefab are assigned in the Inspector
        if (content == null)
        {
            Debug.LogError("[RobotsPanelPresenter] 'content' is not assigned in the Inspector.");
            return;
        }
        if (rowPrefab == null)
        {
            Debug.LogError("[RobotsPanelPresenter] 'rowPrefab' is not assigned in the Inspector.");
            return;
        }

        // Build the list from whatever is already in the directory
        RebuildFromDirectory();

        // Subscribe to directory events so the UI updates live when robots change
        // Guard with a flag so we don't double-subscribe if OnEnable runs again
        if (!_isSubscribed)
        {
            _dir.OnRobotAdded += HandleRobotAdded;
            _dir.OnRobotUpdated += HandleRobotUpdated;
            _dir.OnRobotRemoved += HandleRobotRemoved;
            _isSubscribed = true;
        }
    }

    // Called automatically when this component stops being enabled/visible
    private void OnDisable()
    {
        // If we had a directory and were subscribed, clean up event handlers
        if (_dir != null && _isSubscribed)
        {
            _dir.OnRobotAdded -= HandleRobotAdded;
            _dir.OnRobotUpdated -= HandleRobotUpdated;
            _dir.OnRobotRemoved -= HandleRobotRemoved;
            _isSubscribed = false;
        }
    }

    // Rebuild the whole list from the directory (used on first open)
    private void RebuildFromDirectory()
    {
        // 1) Clear any old rows. (Destroy is delayed until end-of-frame—so we won't reuse rows right now.)
        ClearAllRows();

        // 2) Get a snapshot of robots and sort them (optional).
        List<RobotInfo> robots = new List<RobotInfo>(_dir.GetAll());
        robots.Sort((a, b) => string.Compare(a.Callsign, b.Callsign, StringComparison.OrdinalIgnoreCase));

        // 3) Always create fresh rows during a full rebuild.
        for (int i = 0; i < robots.Count; i++)
        {
            CreateOrUpdateRow(robots[i], allowReuse: false);
        }

        Debug.Log($"[RobotsPanelPresenter:{name}] Rebuild - Rendered {robots.Count} robots.");
    }


    // Remove all child rows from the Content container
    private void ClearAllRows()
    {
        // Loop backwards when destroying children to be safe
        for (int i = content.childCount - 1; i >= 0; i--)
        {
            Transform child = content.GetChild(i);
            Destroy(child.gameObject);
        }
    }

    // === Event handlers from the directory ===

    // A robot was added to the directory (or first seen online)
    private void HandleRobotAdded(RobotInfo r)
    {
        // Create or update the row for this robot
        CreateOrUpdateRow(r);
    }

    // A robot's data changed (name/ip/player/online)
    private void HandleRobotUpdated(RobotInfo r)
    {
        // Create or update the row for this robot
        CreateOrUpdateRow(r);
    }

    // A robot was removed from the directory
    private void HandleRobotRemoved(string robotId)
    {
        // Try to find a row child whose name matches the robotId
        Transform row = content.Find(robotId);

        // If we found it, destroy it
        if (row != null)
        {
            Destroy(row.gameObject);
        }
    }

    // Create a row if it doesn't exist, or update it if it does
    private void CreateOrUpdateRow(RobotInfo r, bool allowReuse = true)
    {
        GameObject rowGO;

        if (allowReuse)
        {
            // Try to find an existing row by name.
            Transform existing = content.Find(r.RobotId);

            if (existing == null)
            {
                // No existing row ? create one.
                rowGO = Instantiate(rowPrefab, content, false);
                rowGO.name = r.RobotId;
            }
            else
            {
                // Reuse the existing row.
                rowGO = existing.gameObject;
            }
        }
        else
        {
            // We are in a full rebuild right after ClearAllRows().
            // Do NOT attempt to reuse rows (they may still be pending-destroy this frame).
            rowGO = Instantiate(rowPrefab, content, false);
            rowGO.name = r.RobotId;
        }

        // ---- Fill UI fields (unchanged) ----
        TextMeshProUGUI nameText = rowGO.transform.Find("Name").GetComponent<TextMeshProUGUI>();
        TextMeshProUGUI ipText = rowGO.transform.Find("Ip").GetComponent<TextMeshProUGUI>();
        TextMeshProUGUI playerText = rowGO.transform.Find("Player").GetComponent<TextMeshProUGUI>();
        Button editButton = rowGO.transform.Find("EditButton").GetComponent<Button>();

        nameText.text = r.Callsign;
        ipText.text = string.IsNullOrEmpty(r.Ip) ? "IP: ?" : $"IP: {r.Ip}";
        playerText.text = string.IsNullOrEmpty(r.AssignedPlayer) ? "Unassigned" : r.AssignedPlayer;

        // (After setting nameText/ipText/playerText)
        editButton.onClick.RemoveAllListeners();                               // Clear old listeners
        editButton.onClick.AddListener(() =>
        {
            // If the popup reference is missing, bail out gracefully.
            if (renamePopup == null)                                           // Popup not wired in inspector?
            {
                Debug.Log("[RobotsPanelPresenter] RenamePopup is not assigned. Skipping edit."); // Helpful log
                return;                                                        // Do nothing
            }

            // Open the popup using the NEW 3-argument callback: (id, newName, playerOrNull).
            renamePopup.Open(
                r.RobotId,                                                     // Robot id we are editing
                r.Callsign,                                                    // Current display name
                                                                               // Pass through the currently assigned player name (null = unassigned).
                                                                               // ALSO treat the legacy literal "Unassigned" as unassigned for safety with older data.
                (string.IsNullOrWhiteSpace(r.AssignedPlayer) || r.AssignedPlayer == "Unassigned")
                    ? null                                                     // Treat blank / "Unassigned" as no player
                    : r.AssignedPlayer,                                        // Otherwise use the stored player name
                (id, newName, newPlayer) =>
                {
                    // Apply the edited name and player assignment back into the directory.
                    _dir.SetCallsign(id, newName);                             // Update callsign in RobotDirectory
                    _dir.SetAssignedPlayer(id, newPlayer);                     // Update or clear player assignment
                }
            );
        });

    }

}
using System.Collections.Generic;      // Lists
using TMPro;                           // TextMeshPro
using UnityEngine;                     // Unity types (MonoBehaviour, GameObject)

public class ServerPanelPresenter : MonoBehaviour
{
    [Header("Robots Number Text")]
    [SerializeField] private GameObject NumRobots;   // Text which displays number of robots

    // === Private fields ===

    private IRobotDirectory _dir;                     // The shared robot registry (from ServiceLocator)
    private bool _isSubscribed = false;               // Tracks if we subscribed to events already (safety)


    private void OnEnable()
    {
        _dir = ServiceLocator.RobotDirectory;


        UpdateRobotsText();

        if (!_isSubscribed)
        {
            _dir.OnRobotAdded += HandleRobotAdded;
            _dir.OnRobotRemoved += HandleRobotRemoved;
            _isSubscribed = true;
        }
    }

    private void OnDisable()
    {
        if (_dir != null && _isSubscribed)
        {
            _dir.OnRobotAdded -= HandleRobotAdded;
            _dir.OnRobotRemoved -= HandleRobotRemoved;
            _isSubscribed = false;
        }
    }

    private void HandleRobotAdded(RobotInfo r)
    {
        UpdateRobotsText();
    }

    private void HandleRobotRemoved(string robotId)
    {
        UpdateRobotsText();
    }

    private void UpdateRobotsText()
    {
        List<RobotInfo> robots = new List<RobotInfo>(_dir.GetAll());
        TextMeshProUGUI NumRobotsText = NumRobots.GetComponent<TextMeshProUGUI>();
        NumRobotsText.text = robots.Count.ToString();
    }
}
